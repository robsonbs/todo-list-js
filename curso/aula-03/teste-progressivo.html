// Validação em tempo real const emailInput = document.getElementById('emailDemo'); const senhaInput = document.getElementById('senhaDemo'); emailInput.addEventListener('input', function() { const email = this.value; const feedback = document.getElementById('feedbackEmail'); if (email.length === 0) { this.className = 'form-control'; feedback.textContent = 'Digite um email...'; feedback.className = 'feedback'; } else if (email.includes('@') && email.includes('.')) { this.className = 'form-control success'; feedback.textContent = '✅ Email válido!'; feedback.className = 'feedback success'; } else { this.className = 'form-control error'; feedback.textContent = '❌ Email inválido'; feedback.className = 'feedback error'; } }); senhaInput.addEventListener('input', function() { const senha = this.value; const feedback = document.getElementById('feedbackSenha'); if (senha.length < 6) { this.className = 'form-control error'; feedback.textContent = '❌ Muito fraca (min 6 caracteres)'; feedback.className = 'feedback error'; } else if (senha.length < 10) { this.className = 'form-control'; feedback.textContent = '⚠️ Média'; feedback.className = 'feedback'; } else { this.className = 'form-control success'; feedback.textContent = '✅ Forte!'; feedback.className = 'feedback success'; } });
